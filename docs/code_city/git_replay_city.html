Folders in the city

Huge folders (mostly data files) start switched off, so the code stays easy to see.

Replay your own project

  1. Copy the script below into a file named export_git_history.py.
  2. Open a terminal in your project and run python export_git_history.py. It writes code_city_history.json.
  3. Click Open history file…, or drag that file onto the city.
The script only reads git history. The page reads the file inside your browser and sends it nowhere.
#!/usr/bin/env python3
"""Export a git repo's history so the Git Replay City page can replay it.

Usage:  python export_git_history.py [repo_folder] [output.json]
Default: the current folder, written to code_city_history.json.
Only needs Python 3 and git. It reads the history; it changes nothing."""
import json, os, re, subprocess, sys

repo = sys.argv[1] if len(sys.argv) > 1 else '.'
out = sys.argv[2] if len(sys.argv) > 2 else 'code_city_history.json'
SKIP = re.compile(r'(^|/)(node_modules|\.git|venv|\.venv|build|dist|\.dart_tool|__pycache__|\.idea|Pods)/|\.(lock|map)$|\.min\.js$|package-lock\.json$')


def git(*a):
    r = subprocess.run(['git', '-c', 'core.quotepath=false', '-C', repo] + list(a), capture_output=True, text=True, encoding='utf-8', errors='replace')
    if r.returncode:
        sys.exit(r.stderr.strip() or 'git failed')
    return r.stdout


def expand(p):
    m = re.match(r'^(.*)\{(.*) => (.*)\}(.*)$', p)
    if m:
        a, b, c, d = m.groups()
        return (a + b + d).replace('//', '/'), (a + c + d).replace('//', '/')
    if ' => ' in p:
        o, n = p.split(' => ', 1)
        return o, n
    return None, p


raw = git('log', '--reverse', '--no-merges', '-M', '--numstat', '--summary', '--pretty=format:@@%h%x09%at%x09%an%x09%s')
commits = []


def finish(c):
    if not c:
        return
    ent = []
    for a, d, p in c['n']:
        old, new = expand(p)
        if a == '-' or SKIP.search(new):
            continue
        a, d = int(a), int(d)
        if old is not None and not SKIP.search(old):
            ent.append(['R', new, a, d, old])
        elif new in c['c']:
            ent.append(['A', new, a, d])
        elif new in c['d']:
            ent.append(['D', new, a, d])
        else:
            ent.append(['M', new, a, d])
    if ent:
        commits.append({'h': c['h'], 't': c['t'], 'a': c['a'], 'm': c['m'], 'f': ent})


cur = None
for line in raw.split('\n'):
    if line.startswith('@@'):
        finish(cur)
        h, t, a, m = (line[2:].split('\t', 3) + ['', '', '', ''])[:4]
        cur = {'h': h, 't': int(t or 0), 'a': a, 'm': m, 'n': [], 'c': set(), 'd': set()}
    elif cur is not None:
        mm = re.match(r'^(\d+|-)\t(\d+|-)\t(.+)$', line)
        if mm:
            cur['n'].append(mm.groups())
        else:
            s = line.strip()
            if s.startswith('create mode'):
                cur['c'].add(s.split(' ', 3)[3])
            elif s.startswith('delete mode'):
                cur['d'].add(s.split(' ', 3)[3])
finish(cur)

name = os.path.basename(os.path.abspath(repo))
branch = git('rev-parse', '--abbrev-ref', 'HEAD').strip()
with open(out, 'w', encoding='utf-8') as fh:
    json.dump({'repo': name, 'branch': branch, 'commits': commits}, fh, separators=(',', ':'))
print('Wrote %s: %d commits, %d bytes' % (out, len(commits), os.path.getsize(out)))
